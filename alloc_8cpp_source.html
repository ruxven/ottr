9struct WorkItem {
10 size_t idx = 0; // index into req.entries
11 long long base = 0; // allocation before remainder distribution
12 long long frac = 0; // distance past the quantum the ideal share fell
13 long long adjusts = 0; // quanta added or removed during remainder distribution
14};

16// Round ideal_numer/denom to a multiple of quantum, per the rounding mode. The
17// division is scaled by the quantum so every mode lands on a quantum boundary
18// rather than rounding to a tick and then snapping.
19long long quantize(long long ideal_numer, long long denom, RoundingMode rounding,
20 long long quantum) {
21 const long long scaled = denom * quantum;
22 long long units = 0;
23 switch (rounding) {
24 case RoundingMode::Round:
25 units = (ideal_numer + scaled / 2) / scaled;
26 break;
27 case RoundingMode::Ceil:
28 units = (ideal_numer + scaled - 1) / scaled;
29 break;
30 case RoundingMode::Truncate:
31 default:
32 units = ideal_numer / scaled;
33 break;
34 }
35 return units * quantum;
36}

38// Rank two candidates for receiving (descending) or surrendering (ascending) a
39// quantum. Returns true when a outranks b. The policy only changes which
40// criteria come first; the trailing criteria keep the order total so the
41// distribution always terminates and stays deterministic.
42bool outranks(const AllocationEntry& ea, const WorkItem& wa,
43 const AllocationEntry& eb, const WorkItem& wb,
44 bool descending, RemainderPolicy policy) {
45 auto prefer = [descending](long long a, long long b) {
46 return descending ? (a > b) : (a < b);
47 };
48
49 if (policy == RemainderPolicy::LargestFrac) {
50 if (wa.frac != wb.frac) return prefer(wa.frac, wb.frac);
51 if (ea.priority != eb.priority) return prefer(ea.priority, eb.priority);
52 } else {
53 if (ea.priority != eb.priority) return prefer(ea.priority, eb.priority);
54 }
55
56 if (ea.remaining_budget != eb.remaining_budget) {
57 return prefer(ea.remaining_budget, eb.remaining_budget);
58 }
59 if (policy != RemainderPolicy::LargestFrac && wa.frac != wb.frac) {
60 return prefer(wa.frac, wb.frac);
61 }
62 // Without this the comparison is identical on every iteration and a single
63 // charge absorbs the whole remainder.
64 if (wa.adjusts != wb.adjusts) {
65 return wa.adjusts < wb.adjusts;
66 }
67 return ea.charge_id < eb.charge_id;
68}

70long long total_weight(const AllocationRequest& req) {
71 long long denom = 0;
72 for (const auto& e : req.entries) {
73 if (e.weight > 0) denom += e.weight;
74 }
75 return denom;

78std::vector<WorkItem> build_work(const AllocationRequest& req, long long denom,
79 RoundingMode rounding, long long quantum) {
80 std::vector<WorkItem> work;
81 work.reserve(req.entries.size());
82 for (size_t i = 0; i < req.entries.size(); ++i) {
83 const AllocationEntry& e = req.entries[i];
84 if (e.weight <= 0) continue;
85 const long long ideal_numer =
86 static_cast<long long>(req.slice_ticks) * static_cast<long long>(e.weight);
87 const long long base = quantize(ideal_numer, denom, rounding, quantum);
88 work.push_back(WorkItem{i, base, ideal_numer % (denom * quantum), 0});
89 }
90 return work;
91}

93// Hand out or claw back whole quanta until the allocation sums to slice_ticks.
94// A slice that is not a multiple of the quantum leaves a sub-quantum tail; it
95// goes to the next eligible charge so the total is always preserved exactly.
96void distribute_ranked(const AllocationRequest& req, std::vector<WorkItem>& work,
97 std::vector<AllocationResultEntry>& out, long long quantum,
98 long long remainder, RemainderPolicy policy) {
99 // Negative remainder means the rounding mode overshot the slice. Only
100 // charges holding at least the amount being taken are eligible, so no
101 // allocation can go negative.
102 auto pick = [&](bool descending, long long needed) -> WorkItem* {
103 WorkItem* best = nullptr;
104 for (WorkItem& w : work) {
105 if (out[w.idx].ticks < needed) continue;
106 if (best == nullptr ||
107 outranks(req.entries[w.idx], w, req.entries[best->idx], *best,
108 descending, policy)) {
109 best = &w;
110 }
111 }
112 return best;
113 };
114
115 while (remainder != 0) {
116 const bool giving = remainder > 0;
117 const long long magnitude = giving ? remainder : -remainder;
118 const long long step = std::min(quantum, magnitude);
119 WorkItem* w = pick(giving, giving ? 0 : step);
120 if (w == nullptr) break;
121 out[w->idx].ticks += giving ? step : -step;
122 w->adjusts += 1;
123 remainder += giving ? -step : step;
124 }
125}

127// Give each eligible charge one quantum in turn, ignoring priority and budget.
128// Subtracting follows the same cycle, skipping charges that cannot cover the
129// step.
130void distribute_round_robin(std::vector<WorkItem>& work,
131 std::vector<AllocationResultEntry>& out,
132 long long quantum, long long remainder) {
133 size_t cursor = 0;
134 while (remainder != 0 && !work.empty()) {
135 const bool giving = remainder > 0;
136 const long long magnitude = giving ? remainder : -remainder;
137 const long long step = std::min(quantum, magnitude);
138
139 bool advanced = false;
140 for (size_t k = 0; k < work.size(); ++k) {
141 const size_t pos = (cursor + k) % work.size();
142 WorkItem& w = work[pos];
143 if (!giving && out[w.idx].ticks < step) continue;
144 out[w.idx].ticks += giving ? step : -step;
145 w.adjusts += 1;
146 remainder += giving ? -step : step;
147 cursor = (pos + 1) % work.size();
148 advanced = true;
149 break;
150 }
151 if (!advanced) break;
152 }
153}

157std::vector<AllocationResultEntry> allocate_weighted(const AllocationRequest& req,
158 RoundingMode rounding,
159 Tick granularity,
160 RemainderPolicy remainder) {
161 std::vector<AllocationResultEntry> out;
162 if (req.slice_ticks <= 0 || req.entries.empty()) return out;
163
164 const long long denom = total_weight(req);
165 if (denom <= 0) return out; // uncharged per spec
166
167 const long long quantum = granularity > 0 ? static_cast<long long>(granularity) : 1;
168
169 out.resize(req.entries.size());
170 for (size_t i = 0; i < req.entries.size(); ++i) {
171 out[i].charge_id = req.entries[i].charge_id;
172 out[i].ticks = 0;
173 }
174
175 std::vector<WorkItem> work = build_work(req, denom, rounding, quantum);
176 for (const WorkItem& w : work) {
177 out[w.idx].ticks += static_cast<Tick>(w.base);
178 }
179
180 long long base_sum = 0;
181 for (const WorkItem& w : work) base_sum += w.base;
182 const long long leftover = static_cast<long long>(req.slice_ticks) - base_sum;
183
184 if (remainder == RemainderPolicy::RoundRobin) {
185 distribute_round_robin(work, out, quantum, leftover);
186 } else {
187 distribute_ranked(req, work, out, quantum, leftover, remainder);
188 }
189
190 return out;
191}
