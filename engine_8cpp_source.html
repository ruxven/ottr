47Tick remaining_budget(const World& world, const RunningTotals& rt,
48 const std::string& charge_id) {
49 auto cit = world.charges.find(charge_id);
50 if (cit == world.charges.end()) return 0;
51 auto it = rt.charged.find(charge_id);
52 Tick charged = (it == rt.charged.end()) ? 0 : it->second;
53 return cit->second.budget_ticks - charged;
54}

63SliceKind classify_ref(const World& world, const Log& start) {
64 if (start.kind == LogKind::None) return SliceKind::Uncharged;
65 if (start.kind == LogKind::Charge) return SliceKind::Direct;
66 // A task-kind log may still name a charge number directly.
67 if (world.charges.count(start.ref_id)) return SliceKind::Direct;
68 auto task_it = world.tasks.find(start.ref_id);
69 if (task_it == world.tasks.end()) return SliceKind::Uncharged;
70 if (task_it->second.weights.empty()) return SliceKind::Uncharged;
71 return SliceKind::Task;
72}

74// Each non-final log opens a slice that ends at the next log.
75std::vector<Slice> day_slices(const World& world, const Day& day) {
76 std::vector<Slice> slices;
77 const auto& logs = day.logs;
78 if (logs.size() < 2) return slices; // validation should have caught
79 slices.reserve(logs.size() - 1);
80 for (size_t i = 0; i + 1 < logs.size(); ++i) {
81 const Log& start = logs[i];
82 Tick ticks = logs[i + 1].time_ticks - start.time_ticks;
83 if (ticks <= 0) continue;
84 SliceKind kind = classify_ref(world, start);
85 if (kind == SliceKind::Uncharged) continue;
86 slices.push_back(Slice{kind, start.ref_id, ticks});
87 }
88 return slices;
89}

91// Merge task slices within a day. Direct charges keep their own slices and are
92// allocated first. Tasks are ordered by first appearance so the allocation
93// sequence follows the file, per SPEC.txt per_day.
94std::vector<Slice> group_day_slices(const std::vector<Slice>& slices) {
95 std::vector<Slice> grouped;
96 std::vector<std::string> task_order;
97 std::unordered_map<std::string, Tick> task_totals;
98 for (const Slice& s : slices) {
99 if (s.kind == SliceKind::Direct) {
100 grouped.push_back(s);
101 continue;
102 }
103 if (task_totals.find(s.ref_id) == task_totals.end()) {
104 task_order.push_back(s.ref_id);
105 }
106 task_totals[s.ref_id] += s.ticks;
107 }
108 grouped.reserve(grouped.size() + task_order.size());
109 for (const std::string& name : task_order) {
110 grouped.push_back(Slice{SliceKind::Task, name, task_totals[name]});
111 }
112 return grouped;
113}

115AllocationRequest build_request(const World& world, const RunningTotals& rt,
116 const Task& task, Tick ticks) {
117 AllocationRequest req;
118 req.slice_ticks = ticks;
119 req.entries.reserve(task.weights.size());
120 for (const auto& w : task.weights) {
121 auto cit = world.charges.find(w.first);
122 if (cit == world.charges.end()) continue; // validator should catch
123 AllocationEntry e;
124 e.charge_id = w.first;
125 e.weight = w.second;
126 e.priority = cit->second.priority;
127 e.remaining_budget = remaining_budget(world, rt, e.charge_id);
128 req.entries.push_back(std::move(e));
129 }
130 // task.weights is unordered; sort so the request is reproducible.
131 std::sort(req.entries.begin(), req.entries.end(),
132 [](const AllocationEntry& a, const AllocationEntry& b) {
133 return a.charge_id < b.charge_id;
134 });
135 return req;
136}

148// Ignore and Warn allocate exactly as before; Warn additionally reports any

150std::vector<AllocationResultEntry> allocate_unbounded(const World& world,
151 const EngineOptions& opts,
152 const Task& task, Tick ticks,
153 const RunningTotals& rt) {

159// Cap and Spill repeatedly allocate what is left across charges that still have
160// budget. Each pass can only reduce the remaining amount, so this terminates;
161// if no charge can take anything, the rest is dropped (Cap) or spilled.
162Tick allocate_budgeted(const World& world, const EngineOptions& opts, const Date& date,
163 const Task& task, Tick ticks, RunningTotals& rt,
164 Aggregation& out) {
165 const CalculationOptions& co = opts.calc_options;
166 Tick remaining = ticks;
167 while (remaining > 0) {
168 AllocationRequest req = build_request(world, rt, task, remaining);
169 req.entries.erase(
170 std::remove_if(req.entries.begin(), req.entries.end(),
171 [](const AllocationEntry& e) { return e.remaining_budget <= 0; }),
172 req.entries.end());
173 if (req.entries.empty()) break;
174
175 auto result = allocate_weighted(req, co.rounding, co.granularity, co.remainder);
176 Tick applied = 0;
177 for (const auto& a : result) {
178 if (a.ticks <= 0) continue;
179 Tick cap = std::max<Tick>(0, remaining_budget(world, rt, a.charge_id));
180 Tick take = std::min<Tick>(a.ticks, cap);
181 if (take <= 0) continue;
182 apply_charge(out, rt, date, "task", task.name, a.charge_id, take, co.audit);
183 applied += take;
184 }
185 if (applied <= 0) break;
186 remaining -= applied;
187 }
188 return remaining;
189}

191void allocate_task(const World& world, const EngineOptions& opts, const Date& date,
192 const Task& task, Tick ticks, RunningTotals& rt, Aggregation& out,
193 std::vector<std::string>* warnings) {
194 const CalculationOptions& co = opts.calc_options;
195
196 if (co.budget == BudgetPolicy::Ignore || co.budget == BudgetPolicy::Warn) {
197 auto result = allocate_unbounded(world, opts, task, ticks, rt);
198 for (const auto& a : result) {
199 if (a.ticks <= 0) continue;
200 if (co.budget == BudgetPolicy::Warn) {
201 Tick cap = remaining_budget(world, rt, a.charge_id);
202 if (a.ticks > cap) warn_overrun(task, a.charge_id, a.ticks - cap, date, warnings);
203 }
204 apply_charge(out, rt, date, "task", task.name, a.charge_id, a.ticks, co.audit);
205 }
206 return;
207 }
208
209 Tick leftover = allocate_budgeted(world, opts, date, task, ticks, rt, out);
210 if (co.budget == BudgetPolicy::Spill && leftover > 0) {
211 apply_charge(out, rt, date, "spill", task.name, co.overflow_charge, leftover, co.audit);
212 }

215void allocate_direct(const World& world, const EngineOptions& opts, const Date& date,
216 const Slice& slice, RunningTotals& rt, Aggregation& out,
217 std::vector<std::string>* warnings) {
218 const CalculationOptions& co = opts.calc_options;
219 Tick capacity = remaining_budget(world, rt, slice.ref_id);
220
221 if (co.budget == BudgetPolicy::Warn && slice.ticks > capacity) {
222 if (warnings != nullptr) {
223 std::ostringstream oss;
224 oss << "warning: charge '" << slice.ref_id << "' over budget by "
225 << hours_text(slice.ticks - capacity) << " hours on " << date.raw
226 << " (direct charge)";
227 warnings->push_back(oss.str());
228 }
229 }
230
231 Tick take = slice.ticks;
232 if (co.budget == BudgetPolicy::Cap || co.budget == BudgetPolicy::Spill) {
233 take = std::min<Tick>(slice.ticks, std::max<Tick>(0, capacity));
234 }
235 apply_charge(out, rt, date, "direct", slice.ref_id, slice.ref_id, take, co.audit);
236
237 Tick leftover = slice.ticks - take;
238 if (co.budget == BudgetPolicy::Spill && leftover > 0) {
239 apply_charge(out, rt, date, "spill", slice.ref_id, co.overflow_charge,
240 leftover, co.audit);
241 }

244void allocate_slice(const World& world, const EngineOptions& opts, const Date& date,
245 const Slice& slice, RunningTotals& rt, Aggregation& out,
246 std::vector<std::string>* warnings) {
247 if (slice.kind == SliceKind::Direct) {
248 allocate_direct(world, opts, date, slice, rt, out, warnings);
249 return;
250 }
251 auto task_it = world.tasks.find(slice.ref_id);
252 if (task_it == world.tasks.end()) return;

256std::vector<const Day*> included_days(const World& world, const DateFilter& filter) {
257 std::vector<const Day*> days;
258 days.reserve(world.days.size());
259 for (const auto& day : world.days) {
260 if (filter.includes(day.date)) days.push_back(&day);
261 }
262 std::sort(days.begin(), days.end(),
263 [](const Day* a, const Day* b) { return a->date.ord < b->date.ord; });
264 return days;

267int month_of(const Date& date) { return date.ord / 100; }

269// PerMonth: accumulate task time across every included day of a month and
270// allocate it at the month's last included day. Direct charges stay on their own
271// day and are allocated as encountered.
272void process_months(const World& world, const EngineOptions& opts,
273 const std::vector<const Day*>& days, RunningTotals& rt,
274 Aggregation& out, std::vector<std::string>* warnings) {
275 size_t i = 0;
276 while (i < days.size()) {
277 const int month = month_of(days[i]->date);
278 std::vector<std::string> task_order;
279 std::unordered_map<std::string, Tick> task_totals;
280 Date last_date = days[i]->date;
281
282 for (; i < days.size() && month_of(days[i]->date) == month; ++i) {
283 last_date = days[i]->date;
284 for (const Slice& s : day_slices(world, *days[i])) {
285 if (s.kind == SliceKind::Direct) {
286 allocate_slice(world, opts, days[i]->date, s, rt, out, warnings);
287 continue;
288 }
289 if (task_totals.find(s.ref_id) == task_totals.end()) {
290 task_order.push_back(s.ref_id);
291 }
292 task_totals[s.ref_id] += s.ticks;
293 }
294 }
295
296 for (const std::string& name : task_order) {
297 auto task_it = world.tasks.find(name);
298 if (task_it == world.tasks.end()) continue;
299 allocate_task(world, opts, last_date, task_it->second,
300 task_totals[name], rt, out, warnings);
301 }
302 }
303}

307bool process_world(const World& world, const EngineOptions& opts, Aggregation& out,
308 [[maybe_unused]] std::string& error_message,
309 std::vector<std::string>* warnings) {
310 out = Aggregation{};
311
312 std::vector<const Day*> days = included_days(world, opts.filter);
313 out.dates_in_order.reserve(days.size());
314 for (const Day* d : days) out.dates_in_order.push_back(d->date);
315
316 RunningTotals rt;
317 if (opts.calc_options.aggregation == AggregationMode::PerMonth) {
318 process_months(world, opts, days, rt, out, warnings);
319 return true;
320 }
321
322 const bool merge_day = opts.calc_options.aggregation == AggregationMode::PerDay;
323 for (const Day* day : days) {
324 std::vector<Slice> slices = day_slices(world, *day);
325 if (merge_day) slices = group_day_slices(slices);
326 for (const Slice& slice : slices) {
327 allocate_slice(world, opts, day->date, slice, rt, out, warnings);
328 }
329 }
330
331 return true;
332}
