24static void compute_widths(const World& world, const Aggregation& agg, std::map<std::string, size_t>& colw, std::vector<std::string>& headers) {
25 headers.clear();
26 headers.push_back("Charge");
27 for (const auto& d : agg.dates_in_order) headers.push_back(d.raw);
28 headers.push_back("Total");
29 headers.push_back("Rem");
30 for (const auto& h : headers) colw[h] = std::max(colw[h], h.size());
31
32 auto ids = sorted_charge_ids(world);
33 for (const auto& cid : ids) {
34 colw["Charge"] = std::max(colw["Charge"], cid.size());
35 Tick total = 0;
36 for (const auto& d : agg.dates_in_order) {
37 Tick v = 0;
38 auto it1 = agg.by_charge_by_day.find(cid);
39 if (it1 != agg.by_charge_by_day.end()) {
40 auto it2 = it1->second.find(d.raw);
41 if (it2 != it1->second.end()) v = it2->second;
42 }
43 auto s = ticks_to_fixed(v);
44 colw[d.raw] = std::max(colw[d.raw], s.size());
45 total += v;
46 }
47 auto st = ticks_to_fixed(total);
48 colw["Total"] = std::max(colw["Total"], st.size());
49 // remaining budget length depends on budget
50 auto cit = world.charges.find(cid);
51 if (cit != world.charges.end()) {
52 Tick rem = cit->second.budget_ticks - total;
53 auto sr = ticks_to_fixed(rem);
54 colw["Rem"] = std::max(colw["Rem"], sr.size());
55 }
56 }
57
58 // Total row widths
59 for (const auto& d : agg.dates_in_order) {
60 auto s = ticks_to_fixed(agg.totals_by_day.count(d.raw) ? agg.totals_by_day.at(d.raw) : 0);
61 colw[d.raw] = std::max(colw[d.raw], s.size());
62 }
63 auto grand = 0ll;
64 for (const auto& kv : agg.totals_by_charge) grand += kv.second;
65 colw["Total"] = std::max(colw["Total"], ticks_to_fixed(grand).size());
66}

80std::string render_full_table(const World& world, const Aggregation& agg) {
81 std::ostringstream oss;
82 std::map<std::string, size_t> colw;
83 std::vector<std::string> headers;
84 compute_widths(world, agg, colw, headers);
85 // Header
86 std::vector<std::pair<std::string, std::string>> hcols;
87 for (const auto& h : headers) hcols.push_back({h, h});
88 append_row(oss, hcols, colw);
89
90 auto ids = sorted_charge_ids(world);
91 for (const auto& cid : ids) {
92 Tick total = 0;
93 std::vector<std::pair<std::string, std::string>> cols;
94 cols.push_back({"Charge", cid});
95 for (const auto& d : agg.dates_in_order) {
96 Tick v = 0;
97 auto it1 = agg.by_charge_by_day.find(cid);
98 if (it1 != agg.by_charge_by_day.end()) {
99 auto it2 = it1->second.find(d.raw);
100 if (it2 != it1->second.end()) v = it2->second;
101 }
102 cols.push_back({d.raw, ticks_to_fixed(v)});
103 total += v;
104 }
105 cols.push_back({"Total", ticks_to_fixed(total)});
106 auto cit = world.charges.find(cid);
107 if (cit != world.charges.end()) {
108 Tick rem = cit->second.budget_ticks - total;
109 cols.push_back({"Rem", ticks_to_fixed(rem)});
110 } else {
111 cols.push_back({"Rem", ""});
112 }
113 append_row(oss, cols, colw);
114 }
115
116 // Total row
117 std::vector<std::pair<std::string, std::string>> tcols;
118 tcols.push_back({"Charge", "Total"});
119 Tick grand = 0;
120 for (const auto& d : agg.dates_in_order) {
121 Tick s = agg.totals_by_day.count(d.raw) ? agg.totals_by_day.at(d.raw) : 0;
122 tcols.push_back({d.raw, ticks_to_fixed(s)});
123 grand += s;
124 }
125 tcols.push_back({"Total", ticks_to_fixed(grand)});
126 tcols.push_back({"Rem", ""});
127 append_row(oss, tcols, colw);
128
129 return oss.str();
130}

132// Trim trailing zeros from a fixed-point number, keeping at least one
133// decimal: "0.800" -> "0.8", "1.000" -> "1.0", "0.000" -> "0.0".
134static std::string trim_zeros(const std::string& s) {
135 if (s.find('.') == std::string::npos) return s + ".0";
136 std::string out = s;
137 while (!out.empty() && out.back() == '0') out.pop_back();
138 if (!out.empty() && out.back() == '.') out.pop_back();
139 if (out.empty() || out == "-") out = "0";
140 if (out.find('.') == std::string::npos) out += ".0";
141 return out;
142}

144// Deltas below this are rounding noise in the display, not real movements.
145static const double kDeltaEpsilon = 0.0005;
146
147// Hours as a signed decimal, e.g. "+0.025" or "-0.1". Zero renders as "0".
148static std::string hours_delta(double hours) {
149 if (hours > -kDeltaEpsilon && hours < kDeltaEpsilon) return "0";
150 std::ostringstream oss;
151 oss.setf(std::ios::fixed);
152 oss << std::setprecision(3) << hours;
153 std::string s = trim_zeros(oss.str());
154 if (hours > 0) return "+" + s;
155 return s;
156}

158// The exact (un-rounded) proportional share of a charge, in hours.
159static std::string hours_exact(long long numer, long long denom) {
160 if (denom <= 0) return "0.0";
161 std::ostringstream oss;
162 oss.setf(std::ios::fixed);
163 oss << std::setprecision(3) << (static_cast<double>(numer) / (static_cast<double>(denom) * 10.0));
164 return trim_zeros(oss.str());
165}

167// The rounding breakdown for one allocation: the exact proportional share,
168// what the rounding mode did to it, what the remainder policy moved, what a
169// budget cap removed, and the net shift against the exact share. Returns an
170// empty string when nothing was adjusted.
171static std::string rounding_note(const AuditEntry& entry) {
172 const RoundingDetail& d = entry.rounding;
173 if (!d.applicable) return "";
174
175 const double exact_ticks = (d.ideal_denom > 0)
176 ? static_cast<double>(d.ideal_numer) / static_cast<double>(d.ideal_denom)
177 : 0.0;
178 const double round_hours = (static_cast<double>(d.rounded_ticks) - exact_ticks) / 10.0;
179 const double rem_hours = static_cast<double>(d.adjust_ticks) / 10.0;
180 const double cap_hours = -static_cast<double>(d.clamped_ticks) / 10.0;
181 const double net_hours = round_hours + rem_hours + cap_hours;
182
183 const bool round_zero = round_hours > -kDeltaEpsilon && round_hours < kDeltaEpsilon;
184 const bool rem_zero = rem_hours > -kDeltaEpsilon && rem_hours < kDeltaEpsilon;
185 const bool cap_zero = cap_hours > -kDeltaEpsilon && cap_hours < kDeltaEpsilon;
186 if (round_zero && rem_zero && cap_zero) return "";
187
188 std::ostringstream oss;
189 oss << " [exact " << hours_exact(d.ideal_numer, d.ideal_denom);
190 if (!round_zero) oss << ", round " << hours_delta(round_hours);
191 if (!rem_zero) oss << ", rem " << hours_delta(rem_hours);
192 if (!cap_zero) oss << ", cap " << hours_delta(cap_hours);
193 oss << ", net " << hours_delta(net_hours) << "]";
194 return oss.str();
195}
